async function handleDownload() { const urlInput = document.getElementById('blog-url'); const statusDiv = document.getElementById('status'); const downloadBtn = document.getElementById('download-btn'); const blogUrl = urlInput.value.trim(); if (!blogUrl) { setStatus('URL을 입력해주세요.', 'error'); return; } setStatus('텍스트를 추출하는 중입니다...', ''); downloadBtn.disabled = true; try { // Netlify Functions 엔드포인트 호출 const response = await fetch('/.netlify/functions/extract', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ url: blogUrl }), }); if (!response.ok) { const errorData = await response.json(); throw new Error(errorData.message || '텍스트 추출에 실패했습니다.'); } const data = await response.json(); const filename = `${data.title || 'blog_content'}.txt`; downloadTxtFile(filename, data.content); setStatus('다운로드가 완료되었습니다.', 'success'); } catch (error) { setStatus(error.message, 'error'); } finally { downloadBtn.disabled = false; } } function downloadTxtFile(filename, text) { const blob = new Blob([text], { type: 'text/plain;charset=utf-8' }); const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = filename; document.body.appendChild(link); link.click(); document.body.removeChild(link); URL.revokeObjectURL(link.href); } function setStatus(message, className) { const statusDiv = document.getElementById('status'); statusDiv.textContent = message; statusDiv.className = className; }
